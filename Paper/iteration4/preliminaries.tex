\section{Preliminaries}

\subsection{Setup and Assumptions}
Let $(Y_i, X_i, Z_i)_{i=1}^{n}$ be an i.i.d. random sample from a joint distribution on $\R^3$ satisfying the structural equation
\begin{equation}
\label{eq:structural}
    Y_i = \beta X_i + \varepsilon_i,
\end{equation}
where $Y_i$ is the variable of interest, $X_i$ is an endogenous regressor, and $Z_i$ is the instrumental variable (IV). Additionally, $\beta \in \R$ is the structural parameter of interest and $\varepsilon_i$ is an unobserved error term. 

For any random quantity $f$, we write $\mu_f := \E[f]$ and $\sigma_f^2 := \Var(f)$ for its mean and variance. All population quantities below are instances of this scheme; for example, $\mu_X = \E[X_i]$, $\mu_{ZX} = \E[Z_i X_i]$, and $\sigma_{ZX}^2 = \Var(Z_i X_i)$.
\todo{MoM can also be in subscript}

We impose the following conditions.
\begin{enumerate}[label=(A\arabic*)]
    \item \label{ass:exog} \textbf{Exogeneity.} $\mu_{Z\varepsilon} = 0$.
    \item \label{ass:relev} \textbf{Relevance.} $\mu_{ZX} \neq 0$.
    \item \label{ass:moments} \textbf{Finite second moments.} $\sigma_{ZX}^2,\ \sigma_{ZY}^2,\ \sigma_{Z\varepsilon}^2 < \infty$.
\end{enumerate}

By \ref{ass:exog} and \ref{ass:relev}, the structural parameter is identified as $\beta = \mu_{ZY}/\mu_{ZX}$.

\begin{remark}[Moment conditions]
The conditions \ref{ass:exog}--\ref{ass:moments} hold for any distribution with finite second moments of the products $Z_iY_i$, $Z_iX_i$, and $Z_i\varepsilon_i$. Thus, no assumptions are imposed on the marginal distributions of $Y_i$, $X_i$, $Z_i$, or $\varepsilon_i$.
\end{remark}

\begin{remark}[Relationship between variances]
\label{rem:minkowski}
The three variances in \ref{ass:moments} are related. Since $Z_i Y_i = \beta Z_i X_i + Z_i\varepsilon_i$, Minkowski's inequality gives
\begin{equation}
\label{eq:minkowski}
    \sigma_{Z\varepsilon} \leq \sigma_{ZY} + \abs{\beta}\sigma_{ZX}.
\end{equation}
To see this, set $U = Z_i Y_i$ and $V = \beta Z_i X_i$, so that $U - V = Z_i\varepsilon_i$. By Minkowski's inequality for $p = 2$,
\[
    \sqrt{\Var(U - V)} \leq \sqrt{\Var(U)} + \sqrt{\Var(V)},
\]
which gives $\sigma_{Z\varepsilon} \leq \sigma_{ZY} + \abs{\beta}\sigma_{ZX}$.
\end{remark}


\subsection{Sub-Gaussian Estimators}
\begin{definition}[Sub-Gaussian Estimators]
\label{def:subgaussian}
Let $(X_i)_{i=1}^{n}$ be a random sample with mean $\mu_X$ and variance $\sigma_X^2$, and let $\hat{\mu}_X$ denote an estimator for $\mu_X$. For some constant $L > 0$, $\hat{\mu}_X$ is said to be a sub-Gaussian estimator for $\mu_X$ for a given confidence level $delta$ if 
\begin{equation}
\label{eq:subgaussian}
    \Prob\!\left[\abs{\hat{\mu}_X - \mu_X} > L\sigma_X \sqrt{\frac{\ln(1/\delta)}{n}}\right] \leq \delta \quad \text{for all } \delta \in (0,1).
\end{equation}
\end{definition}


\subsection{The Median-of-Means Estimator for a Scalar Mean}
\label{sec:mom_prelim}
We present the scalar Median-of-Means (MoM) estimator and its concentration properties. The results in this section are adapted from \cite{devroye2016sub} and \cite{lugosi2019mean}.

\begin{algorithm}[H]
\label{alg:mom}
\caption{Scalar Median-of-Means Estimator}
\KwIn{Sample $(X_i)_{i=1}^{n}$, number of blocks $k$}
\KwOut{Estimate $\hat{\mu}_\mathrm{MoM}$}

Partition $(X_i)_{i=1}^{n}$ into $k$ blocks $B_1, \dots, B_k$ of size $m = \lfloor n/k \rfloor$\;

\For{$j = 1, \dots, k$}{
    Compute the block mean $\hat{\mu}_X^{(j)} = \frac{1}{m}\sum_{i \in B_j} X_i$\;
    }

\Return{$\hat{\mu}_\mathrm{MoM} = \med(\hat{\mu}_X^{(1)}, \dots, \hat{\mu}_X^{(k)})$}\;
\end{algorithm}

\begin{theorem}[MoM concentration]
\label{thm:mom_scalar}
Let $(X_i)_{i=1}^{n}$ be i.i.d.\ with mean $\mu_X$ and variance $\sigma_X^2 < \infty$. Let $k$ be a positive integer, $m = \lfloor n/k \rfloor$, and $\hat{\mu}_{\mathrm{MoM}}$ the MoM estimator of $\mu_X$. Then
\todo{All of the following theorems assume $n=km$}
\begin{equation}
\label{eq:mom_conc}
    \Prob\!\left[\abs{\hat{\mu}_{\mathrm{MoM}} - \mu_X} > \frac{2\sigma_X}{\sqrt{m}}\right] \leq e^{-k/8}.
\end{equation}
Setting $k = \lceil 8\ln(1/\delta) \rceil$ for $\delta \in (0,1)$, this becomes
\begin{equation}
\label{eq:mom_subgauss}
    \Prob\!\left[\abs{\hat{\mu}_{\mathrm{MoM}} - \mu_X} > \sigma_X\sqrt{\frac{32\ln(1/\delta)}{n}}\right] \leq \delta,
\end{equation}
which is sub-Gaussian in the sense of definition~\ref{def:subgaussian} with $L=\sqrt{32}$.
\end{theorem}

\begin{proof}
The proof proceeds in three steps.

\medskip\noindent\textbf{Step 1 (Chebyshev on each block).} Each block mean satisfies $\E[\bar{X}^{(j)}] = \mu_X$ and $\Var(\bar{X}^{(j)}) = \sigma_X^2/m$. By Chebyshev's inequality with threshold $2\sigma_X/\sqrt{m}$,
\[
    \Prob\!\left[\abs{\bar{X}^{(j)} - \mu_X} > \frac{2\sigma_X}{\sqrt{m}}\right] \leq \frac{\sigma_X^2/m}{4\sigma_X^2/m} = \frac{1}{4}.
\]

\medskip\noindent\textbf{Step 2 (Counting argument for the median).} Define $\zeta^{(j)} = \mathbf{1}\{|\bar{X}^{(j)} - \mu_X| > 2\sigma_X/\sqrt{m}\}$. By Step~1, $\E[\zeta^{(j)}] \leq 1/4$. Since the blocks are disjoint subsets of an i.i.d.\ sample, the $(\zeta^{(j)})_{j=1}^{k}$ are independent.

If $|\hat{\mu}_{\mathrm{MoM}} - \mu_X| > 2\sigma_X/\sqrt{m}$, the median exceeds $\mu_X + 2\sigma_X/\sqrt{m}$ or falls below $\mu_X - 2\sigma_X/\sqrt{m}$. In either case, at least $k/2$ of the block means satisfy $|\bar{X}^{(j)} - \mu_X| > 2\sigma_X/\sqrt{m}$, since the median of $k$ numbers can only lie outside an interval if fewer than half the numbers lie inside it. Therefore,
\[
    \Prob\!\left[\abs{\hat{\mu}_{\mathrm{MoM}} - \mu_X} > \frac{2\sigma_X}{\sqrt{m}}\right] \leq \Prob\!\left[\sum_{j=1}^{k} \zeta^{(j)} \geq \frac{k}{2}\right].
\]

\medskip\noindent\textbf{Step 3 (Hoeffding's inequality).} The sum $\sum_{j=1}^{k}\zeta^{(j)}$ consists of $k$ independent random variables with $\zeta^{(j)} \in [0,1]$ and $\E[\sum_j \zeta^{(j)}] \leq k/4$. For $\sum_j \zeta^{(j)}$ to reach $k/2$, it must exceed its mean by at least $k/2 - k/4 = k/4$. By Hoeffding's inequality \citep{hoeffding1963probability},
\[
    \Prob\!\left[\sum_{j=1}^{k} \zeta^{(j)} - \E\!\left[\sum_j \zeta^{(j)}\right] \geq \frac{k}{4}\right] \leq \exp\!\left(\frac{-2(k/4)^2}{k}\right) = e^{-k/8}.
\]
This establishes \eqref{eq:mom_conc}. The sub-Gaussian form \eqref{eq:mom_subgauss} follows by setting $k = \lceil 8\ln(1/\delta)\rceil$.
\end{proof}
