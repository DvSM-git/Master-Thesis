
\section{Median-of-Means Two-Stage Least Squares (MoM-2SLS)}
We first explore the MoM estimator in a univariate setting, with one instrument and one variable.

\subsection{Univariate MoM-2SLS}

In the univariate setting, the 2SLS estimator collapses to the Wald/IV estimator
\[
\hat{\beta}_{IV}=\frac{\sum_{i=1}^{n}Z_iY_i}{\sum_{i=1}^{n}Z_iX_i} = \frac{\bar{S}_{ZY}}{\bar{S}_{ZX}}.
\]
This is a ratio of two sample means. To solve this, we can utilise two MoM-type estimators. The first replaces each sample mean with its MoM counterpart; we call this the ratio-of-medians estimator. The second returns the median of the Wald/IV estimate of each block, we call this the median-of-ratios estimator. 

\subsection{Setup}
Let $(Y_i, X_i, Z_i)_{i=1}^{n}$ be an i.i.d.\ random sample satisfying the structural equation 
\[
    Y_i = \beta X_i + \varepsilon_i,
\]
where $\E[Z_i\varepsilon_i] = 0$ (exogeneity) and $\mu_{ZX} := \E[Z_iX_i] \neq 0$ (relevance). Define
\[
    \mu_{ZY} := \E[Z_iY_i], \quad 
    \sigma_{ZY}^2 := \Var(Z_iY_i), \quad 
    \sigma_{ZX}^2 := \Var(Z_iX_i), \quad 
    \sigma_{Z\varepsilon}^2 := \Var(Z_i\varepsilon_i),
\]
all assumed finite. Both estimators partition the sample into $k$ blocks $B_1, \dots, B_k$ of size $m = \lfloor n/k \rfloor$ and compute block means 
\[
    \hat{S}_{ZY}^{(j)} = \frac{1}{m}\sum_{i \in B_j} Z_iY_i, \qquad \hat{S}_{ZX}^{(j)} = \frac{1}{m}\sum_{i \in B_j} Z_iX_i.
\]
They differ only in how the block-level quantities are aggregated.

\subsection{The Estimators}
 
\begin{definition}[Ratio-of-Medians (RoM)]
\[
    \hat{\beta}_{\mathrm{RoM}} = \frac{\widetilde{S}_{ZY}}{\widetilde{S}_{ZX}}, \qquad \text{where } \widetilde{S}_{ZY} = \med_j(\hat{S}_{ZY}^{(j)}), \quad \widetilde{S}_{ZX} = \med_j(\hat{S}_{ZX}^{(j)}).
\]
The medians are taken separately and the ratio is formed afterwards.
\end{definition}
 
\begin{definition}[Median-of-Ratios (MoR)]
\[
    \hat{\beta}_{\mathrm{MoR}} = \med_j\!\left(\hat{\beta}^{(j)}\right), \qquad \text{where } \hat{\beta}^{(j)} = \frac{\hat{S}_{ZY}^{(j)}}{\hat{S}_{ZX}^{(j)}}.
\]
The ratio is formed within each block and the median is taken over the block-level estimates.
\end{definition}


\subsection{Concentration Results}

\begin{theorem}[Standard IV]
\label{thm:iv_cheb}
Suppose $n \geq 8\sigma_{ZX}^2/(\delta\,\mu_{ZX}^2)$, then
\begin{equation}
\label{eq:iv_bound}
    \Prob\!\left[\abs{\hat{\beta}_{\mathrm{IV}} - \beta} > \frac{2\sqrt{2}\,\sigma_{Z\varepsilon}}{\abs{\mu_{ZX}}\sqrt{\delta\, n}}\right] \leq \delta.
\end{equation}
\end{theorem}

\begin{theorem}[Ratio-of-Medians]
\label{thm:rm}
Let $k = \lceil 8\ln(2/\delta) \rceil$ and $m = \lfloor n/k \rfloor$. Suppose $m \geq 4\sigma_{ZX}^2/\mu_{ZX}^2$. Then
\begin{equation}
\label{eq:rm}
    \Prob\!\left[\abs{\hat{\beta}_{\mathrm{RM}} - \beta} > \frac{2\left(\sigma_{ZY} + \abs{\beta}\sigma_{ZX}\right)}{\abs{\mu_{ZX}}\sqrt{m} - 2\sigma_{ZX}}\right] \leq \delta.
\end{equation}
\end{theorem}
 
\begin{theorem}[Median-of-Ratios]
\label{thm:mr}
Let $k = \lceil 8\ln(1/\delta) \rceil$ and $m = \lfloor n/k \rfloor$. Suppose $m \geq 32\sigma_{ZX}^2/\mu_{ZX}^2$. Then
\begin{equation}
\label{eq:mr}
    \Prob\!\left[\abs{\hat{\beta}_{\mathrm{MR}} - \beta} > \frac{4\sqrt{2}\,\sigma_{Z\varepsilon}}{\abs{\mu_{ZX}}\sqrt{m}}\right] \leq \delta.
\end{equation}
\end{theorem}
The proofs for these results can be found in the tablet notes.

The error bounds can then be written as
\[
    \abs{\hat{\beta} - \beta} \leq C \cdot \frac{\sigma_{\mathrm{eff}}}{\abs{\mu_{ZX}}} \cdot g(\delta, n),
\]
where $C$ is an absolute constant, $\sigma_{\mathrm{eff}}$ is an effective standard deviation, and $g(\delta, n)$ captures the dependence on confidence and sample size. Note that this definition is directly comparable to \refeq{eq:subgaussian}. We utilise these definitions for the following comparisons.

\subsubsection{Effective Variance}
The RoM and IV bound have $\sigma_{ZY} + |\beta|\sigma_{ZX}$ in the numerator while the MoR has $\sigma_{Z\varepsilon}$. Using $Z_iY_i = \beta Z_iX_i + Z_i\varepsilon_i$, it can be shown that
\[
    \sigma_{Z\varepsilon} \leq \sigma_{ZY} + |\beta|\sigma_{ZX}.
\]
Thus, the MoR and IV bounds are tighter. The proof is found in the tablet notes.

The standard IV instrument strength condition dependes inversely on $1/\delta$, meaning that achieving higher confidence requires proportionally more samples.
The MoR and RoM estimators depend logarithmically on $\delta$, which is a significant improvement over the standard IV.

Comparing the two MoM-based estimators, the RoM condition is eight times less stringent than the MoR condition.

\subsubsection{Conclusion for the Univariate Case}

no single estimator dominates. The standard IV has the tightest constants but the weakest instrument strength for moderate $\delta$. Its polynomial tail makes it unsuitable for high-confidence inference. The MoR achieves the tightes sub-Gaussian bound, but requires the strongest instrument condition. The RoM has a mild instrument condition, but a looser effective variance.
